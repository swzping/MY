# 1688一键找同款比价工具 - 实现计划

**Goal:** Build a web tool that helps Pinduoduo sellers find the same product on 1688 by product image or name, compare prices from different suppliers.

**Architecture:** For MVP, this tool helps sellers collect supplier information manually with organized comparison table. Full API integration can be added later.

**Tech Stack:**
- HTML + vanilla JavaScript
- Tailwind CSS via CDN
- Local storage for saving searches

            </div>

// State
let suppliers = [];

// DOM Elements
const productNameInput = document.getElementById('productName');
const pddPriceInput = document.getElementById('pddPrice');
const keywordsInput = document.getElementById('keywords');
const addSupplierBtn = document.getElementById('addSupplierBtn');
const suppliersContainer = document.getElementById('suppliersContainer');
const suppliersBody = document.getElementById('suppliersBody');
const exportCSVBtn = document.getElementById('exportCSVBtn');

// Load saved data from localStorage
document.addEventListener('DOMContentLoaded', () => {
    const saved = localStorage.getItem('1688_finder_data');
    if (saved) {
        try {
            suppliers = JSON.parse(saved);
            renderSuppliers();
            if (suppliers.length > 0) {
                suppliersContainer.classList.remove('hidden');
            }
        } catch (e) {
            console.error('Failed to load saved data');
        }
    }
});

// Add supplier
addSupplierBtn.addEventListener('click', () => {
    const productName = productNameInput.value.trim();
    const pddPrice = parseFloat(pddPriceInput.value);

    if (!productName) {
        showToast('请输入拼多多商品名称');
        return;
    }
    if (isNaN(pddPrice) || pddPrice <= 0) {
        showToast('请输入正确的拼多多售价');
        return;
    }

    // Open modal to add supplier details
    // For simplicity: prompt for supplier info
    const supplierName = prompt('输入供应商名称：');
    if (!supplierName) return;

    const minQuantity = parseInt(prompt('输入起批量：', '1')) || 1;
    const unitPrice = parseFloat(prompt('输入单价：'));
    if (isNaN(unitPrice) || unitPrice <= 0) {
        showToast('请输入正确的单价');
        return;
    }
    const shipping = parseFloat(prompt('输入邮费：', '0')) || 0;

    const profit = pddPrice - unitPrice - shipping;

    const supplier = {
        id: Date.now(),
        productName,
        pddPrice,
        supplierName,
        minQuantity,
        unitPrice,
        shipping,
        profit
    };

    suppliers.push(supplier);
    saveData();
    renderSuppliers();
    suppliersContainer.classList.remove('hidden');
    showToast('供应商已添加');

    // Clear form for next
    // productNameInput.value = '';
});

function saveData() {
    try {
        localStorage.setItem('1688_finder_data', JSON.stringify(suppliers));
    } catch (e) {
        console.error('Failed to save');
    }
}

function renderSuppliers() {
    suppliersBody.innerHTML = '';

    suppliers.forEach(s => {
        const row = document.createElement('tr');
        row.className = 'border-t border-gray-200';

        const profitClass = s.profit > 0 ? 'text-green-600 font-medium' : 'text-red-600 font-medium';

        row.innerHTML = `
            <td class="px-4 py-3">${s.supplierName}</td>
            <td class="px-4 py-3 text-right">${s.minQuantity}</td>
            <td class="px-4 py-3 text-right">${s.unitPrice.toFixed(2)}</td>
            <td class="px-4 py-3 text-right">${s.shipping.toFixed(2)}</td>
            <td class="px-4 py-3 text-right ${profitClass}">${s.profit.toFixed(2)}</td>
            <td class="px-4 py-3 text-center">
                <button data-id="${s.id}" class="delete-btn text-red-600 hover:text-red-800">删除</button>
            </td>
        `;

        // Delete button
        row.querySelector('.delete-btn').addEventListener('click', () => {
            if (confirm('确认删除这个供应商？')) {
                suppliers = suppliers.filter(x => x.id !== s.id);
                saveData();
                renderSuppliers();
                if (suppliers.length === 0) {
                    suppliersContainer.classList.add('hidden');
                }
                showToast('已删除');
            }
        });

        suppliersBody.appendChild(row);
    });
}

// Export CSV
exportCSVBtn.addEventListener('click', () => {
    const headers = ['拼多多商品', '拼多多售价', '供应商', '起批量', '批发价', '邮费', '利润'];
    const rows = [headers.join(',')];

    suppliers.forEach(s => {
        const row = [
            `"${s.productName.replace(/"/g, '""')}"`,
            s.pddPrice,
            `"${s.supplierName.replace(/"/g, '""')}"`,
            s.minQuantity,
            s.unitPrice,
            s.shipping,
            s.profit
        ];
        rows.push(row.join(','));
    });

    const csv = rows.join('\\n');
    const blob = new Blob([csv], { type: 'text/csv' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = '1688-comparison.csv';
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);

    showToast('CSV导出成功');
});
    </script>
</body>
</html>

# 1688找同款比价工具

帮助拼多多卖家整理1688供应商信息，对比不同供应商的价格和利润，方便找货源。

## 功能特点

- 手动添加多个供应商
- 自动计算每件利润
- 表格清晰对比
- 数据保存在浏览器本地
- 导出CSV方便后续分析

## 使用方法

1. 输入拼多多商品名称和售价
2. 逐个添加找到的供应商信息
3. 工具自动计算利润
4. 可以导出CSV保存

## 说明

MVP版本：手动添加供应商信息，适合小批量找货。后续可增加搜索API自动查找。

## 技术

- 纯前端HTML + JavaScript
- localStorage本地存储
- Tailwind CSS via CDN
